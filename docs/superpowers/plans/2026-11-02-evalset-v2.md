```python
"""Pure query hygiene helpers for evalset v2 (no DB, no network)."""

from __future__ import annotations

import csv
import io
import json
import re
import unicodedata
from typing import Any

MAX_SUMMARY_LEN = 500  # app.audit.mcp_access._MAX_SUMMARY_LEN: longer queries are cut off
MIN_QUERY_LEN = 5
MAX_QUERY_LEN = 300

_OTHER_KEYS = ("query", "anchor_value", "external_id", "bucket_id", "path", "slug", "code", "symptom_text")
_OTHER_KEY_RE = re.compile(r"\s(?:%s)=" % "|".join(_OTHER_KEYS))
_ID_ONLY = re.compile(r"^(?:[A-Za-z]+-\d+|[A-Z]{3,}[A-Z0-9_.]*\d[\w.]*|\d{5,})$")
_HINTS = {
    "email": re.compile(r"[\w.+-]+@[\w-]+\.[\w.]+"),
    "ip": re.compile(r"\b\d{1,3}(?:\.\d{1,3}){3}\b"),
    "phone": re.compile(r"\b0\d{1,2}-?\d{3,4}-?\d{4}\b"),
    "url": re.compile(r"https?://"),
    "long_digits": re.compile(r"\d{8,}"),
}
_TOK = re.compile(r"[0-9A-Za-z가-힣]+")
_HANGUL = re.compile(r"[가-힣]")


def norm(q: str) -> str:
    return " ".join(unicodedata.normalize("NFKC", q or "").split()).lower()


def extract_q(summary: str) -> tuple[str, bool]:
    """('query', truncated) from an mcp_access_log.query_summary like 'q=<text>[ key=value…]'."""
    s = (summary or "").strip()
    if not s.startswith("q="):
        return "", False
    truncated = len(s) >= MAX_SUMMARY_LEN
    s = s[2:]
    m = _OTHER_KEY_RE.search(s)
    if m:
        s = s[: m.start()]
    return s.strip(), truncated


def parse_log_tsv(text: str) -> list[dict[str, Any]]:
    """Rows of `date TAB tool TAB count TAB query_summary` (psql \\copy … csv, tab delimiter)."""
    out: list[dict[str, Any]] = []
    for row in csv.reader(io.StringIO(text), delimiter="\t"):
        if len(row) < 4:
            continue
        d, tool, n = row[0], row[1], row[2]
        q, truncated = extract_q("\t".join(row[3:]))
        if not q:
            continue
        try:
            cnt = int(n)
        except ValueError:
            cnt = 1
        out.append({"date": d, "tool": tool, "n": cnt, "q": q, "truncated": truncated})
    return out


def exclusion_reason(q: str, truncated: bool, known: set[str]) -> str | None:
    if truncated:
        return "truncated"
    if len(q) < MIN_QUERY_LEN:
        return "too_short"
    if len(q) > MAX_QUERY_LEN:
        return "too_long"
    if _ID_ONLY.match(q.strip()):
        return "identifier_only"
    if norm(q) in known:
        return "known_case"
    return None


def sensitive_hints(q: str) -> list[str]:
    return [name for name, pat in _HINTS.items() if pat.search(q or "")]


def prepare_real(rows: list[dict[str, Any]], known: set[str]) -> tuple[list[dict[str, Any]], dict[str, int]]:
    stats: dict[str, int] = {"truncated": 0, "too_short": 0, "too_long": 0, "identifier_only": 0, "known_case": 0}
    merged: dict[str, dict[str, Any]] = {}
    for r in rows:
        why = exclusion_reason(r["q"], r["truncated"], known)
        if why:
            stats[why] += 1
            continue
        key = norm(r["q"])
        if key in merged:
            merged[key]["count"] += int(r["n"])
        else:
            merged[key] = {"q": " ".join(r["q"].split()), "count": int(r["n"])}
    ordered = sorted(merged.values(), key=lambda x: (-x["count"], x["q"]))
    items = [
        {"id": f"r-{i:03d}", "q": m["q"], "slice": "real", "source": "real",
         "count": m["count"], "hints": sensitive_hints(m["q"])}
        for i, m in enumerate(ordered, 1)
    ]
    stats["kept"] = len(items)
    return items, stats


def tokens(s: str) -> set[str]:
    return {t.lower() for t in _TOK.findall(unicodedata.normalize("NFKC", s or "")) if len(t) >= 2}


def _tok_match(a: str, b: str) -> bool:
    if a == b:
        return True
    return bool(_HANGUL.search(a + b)) and min(len(a), len(b)) >= 2 and (a.startswith(b) or b.startswith(a))


def title_overlap(query: str, title: str) -> float:
    """Share of query tokens that also occur (Korean-particle tolerant) in the title."""
    qt, tt = tokens(query), tokens(title)
    if not qt:
        return 0.0
    return sum(1 for a in qt if any(_tok_match(a, b) for b in tt)) / len(qt)


def too_easy(query: str, title: str, *, max_ratio: float = 0.5) -> bool:
    return title_overlap(query, title) > max_ratio


def load_known_questions(holdout_path: str | None, evalset_paths: list[str]) -> set[str]:
    known: set[str] = set()
    if holdout_path:
        for it in json.load(open(holdout_path, encoding="utf-8"))["items"]:
            known.add(norm(it["question"]))
    for p in evalset_paths:
        for c in json.load(open(p, encoding="utf-8"))["cases"]:
            if c.get("q"):
                known.add(norm(c["q"]))
    return known
```

```python
def _fake_search(table):
    def search(q, filters, multi_query, top_k):
        return table[multi_query][:top_k]
    return search


def _r(eid, title="t", st="support_history", ee=True):
    return {"external_id": eid, "title": title, "source_type": st, "evidence_eligible": ee}


def test_pool_merges_arms_and_orders_by_agreement():
    table = {
        True: [_r("A#1"), _r("B"), _r("C")],
        False: [_r("B"), _r("D"), _r("A")],
    }
    out = pool.pool_candidates(_fake_search(table), "q", {}, top_n=3, max_pool=10)
    ids = [c["external_id"] for c in out]

    assert set(ids[:2]) == {"A", "B"}
```

```python
"""Tolerant JSON extraction from LLM text (fences, prose before/after)."""

from __future__ import annotations

import json
import re
from typing import Any

_FENCE = re.compile(r"```(?:json)?\s*(.*?)```", re.S)


def extract_json(raw: str | None) -> Any | None:
    text = (raw or "").strip()
    if not text:
        return None
    m = _FENCE.search(text)
    if m:
        text = m.group(1).strip()
    if not text.startswith(("{", "[")):
        starts = [i for i in (text.find("{"), text.find("[")) if i >= 0]
        if not starts:
            return None
        i = min(starts)
        j = max(text.rfind("}"), text.rfind("]"))
        if j <= i:
            return None
        text = text[i : j + 1]
    try:
        return json.loads(text)
    except ValueError:
        return None
```

```python
"""Candidate pooling: union of top-N results from several search settings."""

from __future__ import annotations

import json
import logging
import urllib.request
from collections.abc import Callable
from typing import Any

logger = logging.getLogger("citec.evalv2.pool")

# (arm name, multi_query). Only settings the REST /v1/search exposes.
ARMS: tuple[tuple[str, bool], ...] = (("mq_on", True), ("mq_off", False))

SearchFn = Callable[[str, dict, bool, int], list[dict[str, Any]]]


def bare(eid: str | None) -> str:
    return (eid or "").split("#")[0]


def pool_candidates(
    search: SearchFn,
    q: str,
    filters: dict | None = None,
    *,
    arms: tuple[tuple[str, bool], ...] = ARMS,
    top_n: int = 10,
    max_pool: int = 15,
    extra: list[dict[str, Any]] | None = None,
) -> list[dict[str, Any]]:
    cands: dict[str, dict[str, Any]] = {}
    for name, mq in arms:
        try:
            results = search(q, filters or {}, mq, top_n)
        except Exception as exc:  # noqa: BLE001 — one failed arm must not drop the query
            logger.warning("pool arm %s failed for %r: %s", name, q[:40], exc)
            continue
        seen: set[str] = set()
        for rank, r in enumerate(results[:top_n], 1):
            eid = bare(r.get("external_id"))
            if not eid or eid in seen:
                continue
            seen.add(eid)
            c = cands.setdefault(eid, {
                "external_id": eid, "title": r.get("title") or "",
                "source_type": r.get("source_type"),
                "evidence_eligible": r.get("evidence_eligible"), "arms": {},
            })
            c["arms"][name] = rank
    seeds: dict[str, dict[str, Any]] = {}
    for e in extra or []:
        eid = bare(e.get("external_id"))
        if not eid:
            continue
        c = cands.setdefault(eid, {
            "external_id": eid, "title": e.get("title") or "", "source_type": e.get("source_type"),
            "evidence_eligible": e.get("evidence_eligible"), "arms": {},
        })
        c["arms"]["seed"] = 0
        seeds[eid] = c

    def order_key(c: dict[str, Any]) -> tuple[int, int]:
        ranks = [v for k, v in c["arms"].items() if k != "seed"]
        return (-len(ranks), min(ranks) if ranks else 99)

    ordered = sorted(cands.values(), key=order_key)
    rest = [c for c in ordered if c["external_id"] not in seeds]
    keep_ids = {c["external_id"] for c in rest[: max(0, max_pool - len(seeds))]} | set(seeds)
    return [c for c in ordered if c["external_id"] in keep_ids]


def rest_search(base: str, *, timeout: float = 120.0, attempts: int = 2) -> SearchFn:
    """Read-only POST /v1/search adapter (same request shape the gate harness uses)."""

    def _search(q: str, filters: dict, multi_query: bool, top_k: int) -> list[dict[str, Any]]:
        body = {"q": q, "top_k": top_k, "filters": {"status": "active", **(filters or {})}, "multi_query": multi_query}
        last: Exception | None = None
        for _ in range(attempts):
            try:
                req = urllib.request.Request(
                    base.rstrip("/") + "/v1/search", data=json.dumps(body).encode("utf-8"),
                    headers={"Content-Type": "application/json"}, method="POST",
                )
                with urllib.request.urlopen(req, timeout=timeout) as r:
                    return json.loads(r.read()).get("results", [])
            except Exception as exc:  # noqa: BLE001
                last = exc
        raise RuntimeError(f"search failed: {last}")

    return _search
```

```python
"""LLM-assisted relevance judging. The LLM only *suggests*; evidence quotes are verified in code."""

from __future__ import annotations

import time
import unicodedata
from collections.abc import Callable
from typing import Any

from app.evalv2.llmjson import extract_json

MIN_EVIDENCE_LEN = 8
EXCERPT_CHARS = 1200

RULES = (
    "당신은 사내 기술지원 지식베이스의 검색 평가자입니다. 질의와 후보 문서들을 보고, "
    "각 후보가 질의에 답하는 데 도움이 되는 정도를 판정하세요.\n"
    "등급: 2 = 질의에 직접 답하는 문서, 1 = 관련은 있으나 부분적, 0 = 무관.\n"
    "등급이 1 또는 2이면 해당 후보 본문에서 **글자 그대로 복사한** 근거 문장(8자 이상)을 evidence에 넣으세요. "
    "본문에 없는 문장을 만들거나 요약하지 마세요.\n"
    '반드시 JSON으로만 답하세요: {"judgments":[{"id":"c1","grade":2,"evidence":"…"}]}'
)


def _n(s: str) -> str:
    return " ".join(unicodedata.normalize("NFKC", s or "").split())


def build_messages(query: str, cands: list[dict[str, Any]]) -> list[dict[str, str]]:
    blocks = []
    for i, c in enumerate(cands, 1):
        excerpt = _n(c.get("excerpt") or "")[:EXCERPT_CHARS]
        blocks.append(f"[c{i}] 제목: {_n(c.get('title') or '')}\n본문: {excerpt}")
    content = f"{RULES}\n\n질의: {_n(query)}\n\n후보 문서:\n" + "\n\n".join(blocks)
    return [{"role": "user", "content": content}]


def verify_evidence(evidence: str, body: str) -> bool:
    ev = _n(evidence)
    return len(ev) >= MIN_EVIDENCE_LEN and ev in _n(body)


def _judge_chunk(
    chat: Callable[..., str], query: str, chunk: list[dict[str, Any]],
    bodies: dict[str, str], sleep: Callable[[float], None],
) -> dict[str, dict[str, Any]]:
    raw = ""
    for attempt in range(2):
        raw = chat(build_messages(query, chunk), max_tokens=2500, temperature=0.0, timeout=180.0)
        if (raw or "").strip():
            break
        if attempt == 0:
            sleep(2.0)
    if not (raw or "").strip():
        return {c["external_id"]: {"grade": None, "evidence": "", "status": "empty"} for c in chunk}
    data = extract_json(raw)
    items = data.get("judgments") if isinstance(data, dict) else None
    by_id: dict[str, dict] = {}
    for it in items if isinstance(items, list) else []:
        if isinstance(it, dict) and isinstance(it.get("id"), str):
            by_id[it["id"].strip().strip("[]")] = it
    out: dict[str, dict[str, Any]] = {}
    for i, c in enumerate(chunk, 1):
        eid = c["external_id"]
        it = by_id.get(f"c{i}")
        if it is None:
            out[eid] = {"grade": None, "evidence": "", "status": "missing"}
            continue
        grade = it.get("grade")
        evidence = str(it.get("evidence") or "")
        if isinstance(grade, bool) or grade not in (0, 1, 2):
            out[eid] = {"grade": None, "evidence": "", "status": "invalid"}
        elif grade == 0:
            out[eid] = {"grade": 0, "evidence": "", "status": "ok"}
        elif verify_evidence(evidence, bodies.get(eid, "")):
            out[eid] = {"grade": grade, "evidence": evidence, "status": "ok"}
        else:
            out[eid] = {"grade": None, "evidence": evidence, "status": "unverified"}
    return out


def judge_candidates(
    chat: Callable[..., str], query: str, cands: list[dict[str, Any]], bodies: dict[str, str],
    *, chunk_size: int = 8, sleep: Callable[[float], None] = time.sleep,
) -> dict[str, dict[str, Any]]:
    """external_id -> {grade 0/1/2/None, evidence, status ok|unverified|invalid|missing|empty}."""
    out: dict[str, dict[str, Any]] = {}
    for i in range(0, len(cands), chunk_size):
        out.update(_judge_chunk(chat, query, cands[i : i + chunk_size], bodies, sleep))
    return out
```

```python
from collections.abc import Callable
from typing import Any

from sqlalchemy.orm import Session

from app.evalv2.llmjson import extract_json

MAX_OVERLAP = 0.5

# slice -> source_types to sample from
SLICE_SOURCES: dict[str, tuple[str, ...]] = {
    "pointer": ("confluence_map",),
    "symptom": ("support_history",),

def build_messages(slice_name: str, doc: dict[str, Any], n_queries: int = 2) -> list[dict[str, str]]:
    if slice_name not in _HINT:
        raise ValueError(f"unknown slice: {slice_name}")
    content = (
        "사내 기술지원 지식베이스의 검색 평가용 질의를 만듭니다.\n"
        f"아래 문서를 찾으려는 사람이 입력할 검색 질의 {n_queries}개를 만드세요.\n"
        f"{_HINT[slice_name]}\n"
        "문서 제목의 단어를 그대로 쓰지 마세요. 고객명·사람 이름·IP·계정은 쓰지 마세요.\n"
        '반드시 JSON으로만 답하세요: {"queries":["…","…"]}\n\n'
        f"문서 제목: {doc['title']}\n문서 발췌: {doc.get('excerpt', '')}"
    )
    return [{"role": "user", "content": content}]


def parse_queries(raw: str | None) -> list[str]:
    data = extract_json(raw)
    qs = data.get("queries") if isinstance(data, dict) else None
    if not isinstance(qs, list):
        return []
    return [" ".join(q.split()) for q in qs if isinstance(q, str) and q.strip()]


def accept_queries(queries: list[str], title: str, *, known: set[str], seen: set[str]) -> list[str]:
    out: list[str] = []
    for q in queries:
        n = qf.norm(q)
        if not (MIN_Q <= len(q) <= MAX_Q) or n in known or n in seen:
            continue
        if qf.too_easy(q, title, max_ratio=MAX_OVERLAP):
            continue
        if qf.sensitive_hints(q):
            continue
        seen.add(n)
        out.append(q)
    return out


def sample_docs(session: Session, slice_name: str, n: int, *, seed: str, prefix: str = "") -> list[dict[str, Any]]:
    """Deterministic pseudo-random sample (md5 of external_id + seed) of active docs for a slice."""
    from app.db.models import Document

    sources = SLICE_SOURCES[slice_name]
    stmt = (
        select(Document.external_id, Document.title, Document.body_md, Document.source_type)
        .where(Document.status == "active", Document.source_type.in_(sources))
        .where(func.length(Document.title) >= 8)
        .where(Document.external_id.like(prefix + "%"))
        .order_by(func.md5(Document.external_id + seed))
        .limit(n)
    )
    rows = session.execute(stmt).all()
    return [
        {"external_id": r.external_id, "title": r.title, "source_type": r.source_type,
         "excerpt": " ".join((r.body_md or "").split())[:600]}
        for r in rows
    ]


def generate(
    chat: Callable[..., str], slice_name: str, docs: list[dict[str, Any]], *,
    known: set[str], id_prefix: str, max_per_doc: int = 2,
) -> list[dict[str, Any]]:
    items: list[dict[str, Any]] = []
    seen: set[str] = set()
    for doc in docs:
        try:
            raw = chat(build_messages(slice_name, doc, max_per_doc), max_tokens=800, temperature=0.7, timeout=120.0)
        except Exception as exc:  # noqa: BLE001
            logger.warning("synth LLM failure for %s: %s", doc.get("external_id"), exc)
            continue
        for q in accept_queries(parse_queries(raw), doc["title"], known=known, seen=seen)[:max_per_doc]:
            items.append({
                "id": f"{id_prefix}-{len(items) + 1:03d}", "q": q, "slice": slice_name, "source": "synthetic",
                "seed_doc": {k: doc.get(k) for k in ("external_id", "title", "source_type")},
            })
    return items
```

```python
"""Review-candidate selection, label validation and evalset (dev/frozen) construction."""

from __future__ import annotations

import hashlib
import re
from datetime import datetime, timezone
from typing import Any

from app.evalv2 import qfilter as qf

MARKS = {"O": "expect", "△": "also_ok", "X": None}
CAVEATS = [
    "풀링은 현재 시스템이 올린 문서 위주로 후보를 만들어 절대 recall이 부풀려질 수 있다 — 전/후 상대 비교로 본다.",
    "합성 질의는 제목 겹침률 필터로 편향을 완화했을 뿐 완전히 제거하지 못했다.",
    "LLM 판정은 후보 제안일 뿐이고 최종 판정은 사람의 O/△/X다.",
]


def _pool_rank(c: dict[str, Any]) -> int:
    ranks = [v for k, v in (c.get("arms") or {}).items() if k != "seed"]
    return min(ranks) if ranks else 99


def candidates_for_review(query: dict[str, Any], max_n: int = 8) -> list[dict[str, Any]]:
    """What the reviewer sees: seed doc, LLM positives, items needing a human (unverified/missing/empty),
    and the pool's top-3 regardless of the LLM. Capped; pool order is kept."""
    cands = query["candidates"]
    top3 = {c["external_id"] for c in sorted(cands, key=_pool_rank)[:3] if _pool_rank(c) < 99}

    def priority(c: dict[str, Any]) -> int:
        llm = c.get("llm") or {}
        if "seed" in (c.get("arms") or {}):
            return 0
        if llm.get("grade") in (1, 2):
            return 1
        if llm.get("status") in ("unverified", "missing", "empty", "invalid"):
            return 2
        if c["external_id"] in top3:
            return 3
        return 9

    picked = [c for c in cands if priority(c) < 9]
    picked.sort(key=lambda c: (priority(c), _pool_rank(c)))
    keep = {c["external_id"] for c in picked[:max_n]}
    return [c for c in cands if c["external_id"] in keep]


def validate_review(judged: dict[str, Any], review: dict[str, Any]) -> list[str]:
    errs: list[str] = []
    if not str(review.get("reviewer") or "").strip():
        errs.append("reviewer is empty")
    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", str(review.get("reviewed_on") or "")):
        errs.append("reviewed_on must be YYYY-MM-DD")
    qmap = {q["id"]: {c["external_id"] for c in q["candidates"]} for q in judged["queries"]}
    for qid, marks in (review.get("labels") or {}).items():
        if qid not in qmap:
            errs.append(f"unknown query id {qid}")
            continue
        for eid, mark in (marks or {}).items():
            if eid not in qmap[qid]:
                errs.append(f"{qid}: unknown candidate {eid}")
            if mark not in MARKS:
                errs.append(f"{qid}/{eid}: bad mark {mark!r}")
    return errs


def _split(q: str) -> str:
    return "frozen" if int(hashlib.sha1(qf.norm(q).encode("utf-8")).hexdigest(), 16) % 2 else "dev"


def build_evalsets(judged: dict[str, Any], review: dict[str, Any], *, evalset_id: str) -> dict[str, Any]:
    errs = validate_review(judged, review)
    if errs:
        raise ValueError("invalid review: " + "; ".join(errs))
    now = datetime.now(timezone.utc).isoformat(timespec="seconds")
    cases = {"dev": [], "frozen": []}
    report = {"kept": 0, "no_answer": 0, "unreviewed": 0}
    for q in judged["queries"]:
        marks = (review.get("labels") or {}).get(q["id"])
        if not marks:
            report["unreviewed"] += 1
            continue
        expect = [e for e in (c["external_id"] for c in q["candidates"]) if MARKS.get(marks.get(e)) == "expect"]
        also = [e for e in (c["external_id"] for c in q["candidates"]) if MARKS.get(marks.get(e)) == "also_ok"]
        if not expect:
            report["no_answer"] += 1
            continue
        split = _split(q["q"])
        cases[split].append({
            "id": q["id"], "group": f"v2_{q['slice']}", "kind": "search", "q": q["q"], "filters": {},
            "expect": expect, "also_ok": also, "k": 10, "source": q["source"], "split": split,
            "reviewed_on": review["reviewed_on"],
        })
        report["kept"] += 1
    base = {
        "provenance": f"실사용 질의(운영 mcp_access_log) + 합성 질의, 풀링 + LLM 보조 판정, 검수자 {review['reviewer']} "
                      f"({review['reviewed_on']}). 모델: {judged.get('model')}",
        "caveats": CAVEATS,
    }
    return {
        "dev": {"evalset_id": f"{evalset_id}-dev", **base, "cases": cases["dev"]},
        "frozen": {"evalset_id": f"{evalset_id}-frozen", **base, "frozen_at_utc": now, "cases": cases["frozen"]},
        "report": report,
    }
```

```python
def paired_bootstrap(before: list[float], after: list[float], iters: int = BOOT_ITERS,
                     seed: int = 20261102) -> tuple[float, float, float]:
    """(평균 차이 after-before, 95% 하한, 95% 상한) — 같은 케이스끼리 짝지어 재표집한다."""
    n = len(before)
    if n == 0 or n != len(after):
        raise ValueError("paired_bootstrap needs equal, non-empty inputs")
    diffs = [a - b for a, b in zip(after, before)]
    mean = sum(diffs) / n
    rng = random.Random(seed)
    means = sorted(sum(diffs[rng.randrange(n)] for _ in range(n)) / n for _ in range(iters))
    return mean, means[int(0.025 * iters)], means[int(0.975 * iters) - 1]
```

```python
        print("\n== v2 군 (통계 판정: recall@10 차이의 95% 구간이 전부 음수일 때만 실패) ==")
        print(f"{'군':<18}{'n':>4}{'metric':>12}{'Δ':>8}{'95% 구간':>20}  판정")
    for grp in stat_groups:
        ids = [i for i in cc if i in bc and cc[i]["group"] == grp]
        if len(ids) < MIN_STAT_N:
            print(f"{grp:<18}{len(ids):>4}  표본 부족(n<{MIN_STAT_N}) — 판정하지 않음")
            continue
        for name, fn in (("recall@10", _hit10), ("mrr", _rr)):
            mean, lo, hi = paired_bootstrap([fn(bc[i]) for i in ids], [fn(cc[i]) for i in ids])
            verdict = "차이 불분명"
            if hi < 0:
                verdict = "유의한 하락"
            elif lo > 0:
                verdict = "유의한 개선"
            print(f"{grp:<18}{len(ids):>4}{name:>12}{mean:>+8.3f}{f'[{lo:+.3f}, {hi:+.3f}]':>20}  {verdict}")
            if name == "recall@10" and hi < 0:
                failed.append(f"{grp} recall@10 유의한 하락 (Δ={mean:+.3f}, 95%=[{lo:+.3f}, {hi:+.3f}])")
    if flips:
        print(f"\nv2 군 단건 pass→fail(참고, 판정에 쓰지 않음): {len(flips)}건")
```

evalset 군별 표(`== evalset (군별 통과 수) ==`)는 그대로 둔다. 모듈 docstring의 "게이트(compare)" 설명 끝에 한 줄 추가: `v2_ 접두사 군은 케이스 단위 회귀 대신 recall@10 차이의 페어 부트스트랩 95% 구간으로 판정한다.`
